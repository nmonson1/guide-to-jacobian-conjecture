\[
\widehat{\mathcal O}_{K_{3,7},G}

\C[[t_1,\ldots,t_{11}]]

\[
\operatorname{length}\mathcal R=584.
\]

\[
[\operatorname{Spf}\mathcal R/\Gm].
\]
